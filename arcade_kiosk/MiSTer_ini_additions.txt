; In the existing [MiSTer] section:
fb_terminal=1          ; the kiosk draws on the menu core's terminal screen
recents=0              ; nothing gets written to the SD card on each launch
; Monitor turned on its side (screen = tate_cw / tate_ccw in kiosk.ini)?
; Rotate the MiSTer's own menu too, so it's readable after unlocking:
; 1 = rotate right, 2 = rotate left. Use whichever matches the kiosk.
;osd_rotate=1
; Leave osd_lock unset. The kiosk does the locking (completely, with its
; kiosk build of the MiSTer main program), and with stock MiSTer osd_lock
; would stop the kiosk saving NVRAM for games like NBA Jam.

; Add this section at the end of the file. It makes the menu core (and so the
; game list) output 320x240 at 15 kHz through the analog port.
[menu]
vga_scaler=1
video_mode=320,8,32,24,240,4,3,16,6048
; composite_sync=1     ; uncomment for SCART / JAMMA / monitors needing CSync
